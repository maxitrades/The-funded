📋 Nav Auth Snippet — add to index.html, challenges.html, etc.

Step 1: Make sure your nav has these IDs on the links:

<!-- In your nav, add these IDs -->
<a href="login.html"    id="navLogin"    class="...">Login</a>
<a href="signup.html"   id="navSignup"   class="...">Sign Up</a>
<a href="dashboard.html" id="navDashboard" class="..." style="display:none">Dashboard</a>
<a href="profile.html"  id="navProfile"  class="..." style="display:none">My Profile</a>

Step 2: Paste this script at the bottom of every page (before </body>):

<script type="module">
  import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";
  import { getAuth, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-auth.js";

  const firebaseConfig = {
    apiKey: "AIzaSyDaKGU5Nrtof0lA7GwZN9baIwdpVZaftbQ",
    authDomain: "the-funded.firebaseapp.com",
    projectId: "the-funded",
    storageBucket: "the-funded.firebasestorage.app",
    messagingSenderId: "656425279116",
    appId: "1:656425279116:web:d309fc0a731066f28225a9"
  };
  const app = initializeApp(firebaseConfig);
  const auth = getAuth(app);

  onAuthStateChanged(auth, (user) => {
    const login    = document.getElementById('navLogin');
    const signup   = document.getElementById('navSignup');
    const dash     = document.getElementById('navDashboard');
    const profile  = document.getElementById('navProfile');

    if (user) {
      // Logged in: hide login/signup, show dashboard/profile
      if (login)   login.style.display   = 'none';
      if (signup)  signup.style.display  = 'none';
      if (dash)    dash.style.display    = '';
      if (profile) profile.style.display = '';
    } else {
      // Logged out: show login/signup, hide dashboard/profile
      if (login)   login.style.display   = '';
      if (signup)  signup.style.display  = '';
      if (dash)    dash.style.display    = 'none';
      if (profile) profile.style.display = 'none';
    }
  });
</script>

That's it! This handles all public-facing pages. Dashboard and Profile pages have their own full auth logic.
